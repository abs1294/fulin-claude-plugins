```bash
#!/usr/bin/env bash
# 真實端到端：用 claude -p 無人值守跑一次完整 /harness:init（以 --plugin-dir 載入開發中的 harness），跑完對產出跑 init-verify。
# 用法：bash run-e2e.sh <工作目錄（系統暫存目錄底下，不在 repo 內）>
set -u
WORK="$1"
PLUGIN='C:\Users\User\Claude\fulin-claude-plugins\plugins\harness'
REPO_PLUGIN=/c/Users/User/Claude/fulin-claude-plugins/plugins/harness
mkdir -p "$WORK"
PROJ="$WORK/shipping-portal"
node "$(cygpath -w "$REPO_PLUGIN/tests/lib/fixtures.js")" single-web "$(cygpath -w "$PROJ")" || exit 2
cd "$PROJ" || exit 2
START=$(date +%s)
echo "start $(date -Iseconds)" > "$WORK/timing.txt"
PROMPT='/harness:init 無人值守執行（headless）：沒有人會回答問題。每一題照你的推薦選項代決，用 init-flow.js answer 寫進答案檔並標 "delegated":true；需要安裝外部 plugin 或工具（git-commit、Codex、archify 以外的東西）的一律選不裝並在收尾回報寫明；照 SKILL.md 的硬規則與每個 Phase 的細節檔做完全部 Phase，最後跑 init-flow.js done。'
env -u CLAUDECODE -u CLAUDE_CODE_ENTRYPOINT MSYS_NO_PATHCONV=1 \
  claude -p "$PROMPT" --plugin-dir "$PLUGIN" --permission-mode bypassPermissions \
  --max-budget-usd 60 --output-format stream-json --verbose \
  > "$WORK/stream.jsonl" 2> "$WORK/stderr.txt"
CODE=$?
END=$(date +%s)
echo "end $(date -Iseconds) exit=$CODE seconds=$((END-START))" >> "$WORK/timing.txt"
node "$(cygpath -w "$REPO_PLUGIN/skills/init/scripts/init-verify.js")" "$(cygpath -w "$PROJ")" > "$WORK/init-verify.txt" 2>&1
echo "init-verify exit=$?" >> "$WORK/init-verify.txt"
node "$(cygpath -w "$REPO_PLUGIN/skills/init/scripts/init-flow.js")" status "$(cygpath -w "$PROJ")" > "$WORK/init-flow-status.txt" 2>&1
exit $CODE
```

spawn 前清掉 `CLAUDECODE`（`env -u`），並設 `MSYS_NO_PATHCONV=1`（否則 Git Bash 會把 `/harness:init` 改寫成路徑）。
